        "import torch\n",
        "import torch.nn as nn\n",
        "import torch.nn.functional as F\n",
        "from torchvision import transforms, models\n",

        "# ========== CONFIGURATION ==========\n",
        "IMG_SIZE = 224\n",
        "DEVICE = torch.device(\"cuda\" if torch.cuda.is_available() else \"cpu\")\n",
        "MODEL_PATH = \"/kaggle/input/www/other/default/1/skin_disease_mtl_best.pth\"\n",
        "\n",
        "TARGET_CLASSES = [\n",
        "    'nevus', 'melanoma', 'basal cell carcinoma', 'seborrheic keratosis',\n",
        "    'squamous cell carcinoma', 'actinic keratosis', 'pigmented benign keratosis'\n",
        "]\n",
        "\n",
        "# ========== MTL MODEL ARCHITECTURE ==========\n",
        "class AIDermatologist(nn.Module):\n",
        "    def __init__(self, num_classes):\n",
        "        super().__init__()\n",
        "        base = models.convnext_tiny(weights=None)\n",
        "        self.features = base.features\n",
        "        self.pool = nn.AdaptiveAvgPool2d(1)\n",
        "        self.type_head = nn.Linear(768, num_classes)\n",
        "        self.malignant_head = nn.Linear(768, 1)\n",
        "\n",
        "    def forward(self, x):\n",
        "        f = self.pool(self.features(x)).flatten(1)\n",
        "        return self.type_head(f), self.malignant_head(f)\n",
        "\n",
        "# ========== GLOBAL MODEL LOADING ==========\n",
        "def load_trained_model():\n",
        "    global TARGET_CLASSES\n",
        "    model = AIDermatologist(len(TARGET_CLASSES))\n",
        "    try:\n",
        "        checkpoint = torch.load(MODEL_PATH, map_location=DEVICE, weights_only=False)\n",
        "        if isinstance(checkpoint, dict) and 'state_dict' in checkpoint:\n",
        "            state_dict = checkpoint['state_dict']\n",
        "            if 'names' in checkpoint:\n",
        "                TARGET_CLASSES = checkpoint['names']\n",
        "                model.type_head = nn.Linear(768, len(TARGET_CLASSES))\n",
        "        else:\n",
        "            state_dict = checkpoint\n",
        "        model.load_state_dict(state_dict)\n",
        "    except Exception: pass\n",
        "\n",
        "    model = model.to(DEVICE).eval()\n",
        "    base_transform = transforms.Compose([\n",
        "        transforms.Resize((IMG_SIZE, IMG_SIZE)),\n",
        "        transforms.ToTensor(),\n",
        "        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])\n",
        "    ])\n",
        "    return model, base_transform\n",
        "\n",
        "GLOBAL_MODEL, GLOBAL_TRANSFORM = load_trained_model()\n",
        "\n",
        "# ========== STAGE 1: THE HARD GATE (Computer Vision) ==========\n",
        "def clinical_validation(img_np):\n",
        "    img_pil = Image.fromarray(img_np).convert('YCbCr')\n",
        "    ycbcr = np.array(img_pil)\n",
        "    cb, cr = ycbcr[:,:,1], ycbcr[:,:,2]\n",
        "    skin_mask = (cb > 80) & (cb < 125) & (cr > 135) & (cr < 175)\n",
        "    skin_ratio = np.sum(skin_mask) / (img_np.shape[0] * img_np.shape[1])\n",
        "    img_gray = Image.fromarray(img_np).convert('L')\n",
        "    std_dev = np.std(np.array(img_gray))\n",
        "    if skin_ratio < 0.25 or std_dev > 85 or std_dev < 5:\n",
        "        return False\n",
        "    return True\n",
        "\n",
        "# ========== PREDICTION LOGIC ==========\n",
        "def predict_pro(img):\n",
        "    if img is None: return None, \"Waiting for image...\", \"\"\n",
        "    if not clinical_validation(img):\n",
        "        return None, \"INVALID IMAGE\", \"### ❌ Error: Non-Clinical Image\\n\\nThe system has detected this is not a medical-grade skin sample. Please provide a clear, focused photo of the skin area.\"\n",
        "\n",
        "    img_pil = Image.fromarray(img).convert('RGB')\n",
        "    input_tensor = GLOBAL_TRANSFORM(img_pil).unsqueeze(0).to(DEVICE)\n",
        "    with torch.no_grad():\n",
        "        type_logits, mal_logits = GLOBAL_MODEL(input_tensor)\n",
        "        probs = F.softmax(type_logits, dim=1).cpu().numpy()[0]\n",
        "        mal_prob = torch.sigmoid(mal_logits).item()\n",
        "\n",
        "    confidences = {TARGET_CLASSES[i]: float(probs[i]) for i in range(len(TARGET_CLASSES))}\n",
        "    top_label = max(confidences, key=confidences.get)\n",
        "    top_conf = confidences[top_label] * 100\n",
        "    if top_conf < 50.0:\n",
        "        return None, \"UNCERTAIN / INVALID\", \"### ❌ Error: Unrecognized Pattern\\n\\nThe AI cannot confidently identify a disease. Ensure the lesion is centered and the lighting is natural.\"\n",
        "\n",
        "    timestamp = datetime.now().strftime(\"%Y-%m-%d %H:%M:%S\")\n",
        "    report = f\"### 📄 Analysis Report\\n**Timestamp:** {timestamp}\\n**Primary Finding:** {top_label.upper()}\\n\"\n",
        "    report += f\"**Confidence Score:** {top_conf:.2f}%\\n\"\n",
        "    report += f\"**Malignancy Risk:** {('HIGH' if mal_prob > 0.5 else 'LOW')} ({mal_prob*100:.1f}%)\\n\\n\"\n",
        "    report += \"> *Note: This result is based on digital pattern recognition.*\"\n",
        "    return confidences, f\"Detection: {top_label.title()}\", report\n",
